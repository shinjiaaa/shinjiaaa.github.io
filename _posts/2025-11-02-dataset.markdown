설치가 완료되었으면 파이토치 데이터셋을 정의해 줍니다.

import torch
from torch.utils.data import Dataset
import decord
import numpy as np
import torchvision.transforms as T

decord.bridge.set_bridge('torch')

class NexarDataset(Dataset):
def **init**(self, hf_dataset, num_frames=16, resize=(224,224)):
self.dataset = hf_dataset
self.num_frames = num_frames
self.resize = resize
self.transform = T.Compose([
T.Resize(resize),
T.ConvertImageDtype(torch.float32),
T.Normalize(mean=[0.485,0.456,0.406],
std=[0.229,0.224,0.225])
])

    def __len__(self):
        return len(self.dataset)

    def __getitem__(self, idx):
        item = self.dataset[idx]
        video_path = item["video_path"]
        label = torch.tensor(item["label"], dtype=torch.long)

        vr = decord.VideoReader(video_path)
        total_frames = len(vr)
        indices = np.linspace(0, total_frames-1, self.num_frames, dtype=int)
        frames = vr.get_batch(indices)  # (T,H,W,C)
        frames = frames.permute(0,3,1,2) # (T,C,H,W)
        frames = self.transform(frames)
        return frames, label

이제 모델 정의와 트레인 코드를 실행해 봅시다.

import torch.nn as nn
import torchvision.models as models
from torch.utils.data import DataLoader
from tqdm import tqdm

train_ds = NexarDataset(dataset)
train_loader = DataLoader(train_ds, batch_size=2, shuffle=True)

class CollisionClassifier(nn.Module):
def **init**(self):
super().**init**()
base = models.resnet18(weights=models.ResNet18_Weights.DEFAULT)
base.fc = nn.Identity()
self.backbone = base
self.fc = nn.Linear(512, 2)

    def forward(self, x):
        B, T, C, H, W = x.shape
        x = x.view(B*T, C, H, W)
        feats = self.backbone(x)
        feats = feats.view(B, T, 512)
        feats = feats.mean(dim=1)
        logits = self.fc(feats)
        return logits

device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
model = CollisionClassifier().to(device)
criterion = nn.CrossEntropyLoss()
optimizer = torch.optim.Adam(model.parameters(), lr=1e-4)
def train_one_epoch(model, dataloader, optimizer, criterion, device):
model.train()
total_loss, total_correct = 0,0
for frames, labels in tqdm(dataloader):
frames, labels = frames.to(device), labels.to(device)
logits = model(frames)
loss = criterion(logits, labels)

        optimizer.zero_grad()
        loss.backward()
        optimizer.step()

        total_loss += loss.item() * frames.size(0)
        total_correct += (logits.argmax(1) == labels).sum().item()

    return total_loss/len(dataloader.dataset), total_correct/len(dataloader.dataset)

EPOCHS = 5
for epoch in range(EPOCHS):
loss, acc = train_one_epoch(model, train_loader, optimizer, criterion, device)
print(f"Epoch {epoch+1}/{EPOCHS} | Loss: {loss:.4f} | Acc: {acc:.4f}")
